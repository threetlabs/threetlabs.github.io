Install Prerequisites
One-time setup for all tools in the pipeline
# Install Go tools
go install -v github.com/projectdiscovery/subfinder/v2/cmd/subfinder@latest
go install -v github.com/projectdiscovery/httpx/cmd/httpx@latest
go install -v github.com/projectdiscovery/dnsx/cmd/dnsx@latest

# Install Python tools
pip3 install censys wafw00f

# Verify installs
subfinder -version && httpx -version

WhoXY Reverse WHOIS (also done in-browser above)
Find all domains registered by a company — pipe to file
# WhoXY reverse WHOIS — company name search
curl -s "https://api.whoxy.com/?key=YOUR_KEY&reverse=whois&company=TARGET_COMPANY&mode=domains&page=1" \
  | jq -r '.search_result[].domain_name' \
  | tee root_domains.txt

# Paginate all results (WhoXY returns 20 per page)
for i in $(seq 1 10); do
  curl -s "https://api.whoxy.com/?key=YOUR_KEY&reverse=whois&company=TARGET_COMPANY&mode=domains&page=$i" \
    | jq -r '.search_result[].domain_name' >> root_domains.txt
done
sort -u root_domains.txt -o root_domains.txt
2
Certificate Transparency (crt.sh)
Discover additional domains via TLS certificate logs
# Pull all domains matching company from crt.sh
while IFS= read -r domain; do
  curl -s "https://crt.sh/?q=%25.$domain&output=json" \
    | jq -r '.[].name_value' \
    | sed 's/\*\.//g' \
    | sort -u
done < root_domains.txt | sort -u >> all_domains.txt
cat root_domains.txt all_domains.txt | sort -u > master_domains.txt
3
Subfinder — Passive Subdomain Enumeration
Runs 40+ passive sources: VirusTotal, Shodan, Censys, etc.
# Run subfinder against all discovered root domains
subfinder \
  -dL master_domains.txt \
  -all \
  -recursive \
  -silent \
  -o subdomains_raw.txt

# Optional: also run chaos (ProjectDiscovery dataset)
while IFS= read -r domain; do
  chaos -d $domain -silent >> subdomains_raw.txt
done < master_domains.txt

sort -u subdomains_raw.txt -o subdomains.txt
wc -l subdomains.txt  # total unique subdomains
4
DNS Resolution (dnsx)
Resolve only live subdomains before probing — saves time
# Resolve subdomains, keep only those with DNS records
dnsx \
  -l subdomains.txt \
  -resp \
  -a -cname \
  -threads 100 \
  -silent \
  -o resolved.txt

# Extract just hostnames for httpx
awk '{print $1}' resolved.txt > live_hosts.txt
5
httpx — HTTP Probe + Title + Tech Detection
Probe all live hosts for status, title, server, and tech stack
# Full httpx probe with tech detection (Wappalyzer-powered)
httpx \
  -l live_hosts.txt \
  -title \
  -status-code \
  -tech-detect \
  -server \
  -content-length \
  -follow-redirects \
  -threads 50 \
  -rate-limit 150 \
  -timeout 10 \
  -json \
  -o httpx_results.json

# Human-readable summary (live 2xx hosts only)
cat httpx_results.json | jq -r \
  'select(.status_code >= 200 and .status_code < 300) | "\(.url) [\(.status_code)] \(.title)"' \
  | tee live_results.txt
